export const azul = '#0050ff'
export const tinta = '#00001e'
export const gris = '#3c3c46'
export const suave = '#f3f3f5'
export const linea = '#e3e3e8'
export const verde = '#00805a'
export const ambar = '#7d6a00'

export const ESTADO = (() => {
  try {
    return JSON.parse(import.meta.env.STORYBOOK_STARTERSLUG_ESTADO)
  } catch {
    return null
  }
})()

export const piezas = ESTADO?.piezas ?? []
export const resumen = ESTADO?.resumen ?? {}
export const nuevas = piezas.filter((p) => p.cambio === 'nuevo')
export const modificadas = piezas.filter((p) => p.cambio === 'modificado')
export const enCurso = piezas.filter((p) => p.enCurso)

{/*
  EL ROTULO «EN CURSO» NO VIAJA A PRO: esa marca es
  para nosotros —local y pre—, no para quien recibe la entrega.

  Se apaga lo que se PINTA, no el dato: las piezas siguen viajando y siguen siendo
  las mismas. Por eso aqui desaparecen la pastilla, su fila de la leyenda, la cifra
  y el bloque del final — si se dejara el bloque con la lista vacia diria «nada en
  curso», que es justo lo contrario de la verdad.

  El destino lo pone `deploy.mjs` y lo reparte `main.js`. En local va vacio, asi
  que se ve todo.
*/}
export const aPro = import.meta.env.STORYBOOK_STARTERSLUG_DESTINO === 'entrega'

export const fechaLarga = (iso) => {
  if (!iso) return '—'
  const d = new Date(iso)
  return d.toLocaleDateString('es-ES', { day: 'numeric', month: 'long', year: 'numeric' })
}

{/* Mismos colores y mismo reparto de peso que `renderLabel` en manager.jsx: sólido
    lo nuevo, suave lo modificado. La leyenda tiene que parecerse a lo que hay en la
    barra lateral, o no es una leyenda. */}

export const Marca = ({ tipo }) => {
  const m = {
    nuevo: { c: '#fff', f: verde, s: '+' },
    modificado: { c: azul, f: '#dbe6ff', s: '~' },
  }[tipo]
  if (!m) return null
  return (
    <span
      style={{
        display: 'inline-flex',
        alignItems: 'center',
        justifyContent: 'center',
        width: '18px',
        height: '18px',
        borderRadius: '50%',
        background: m.f,
        color: m.c,
        fontSize: '12px',
        fontWeight: 700,
        flex: '0 0 auto',
      }}
    >
      {m.s}
    </span>
  )
}

export const Pastilla = () => (
  <span
    style={{
      padding: '2px 7px',
      borderRadius: '4px',
      background: '#fbf4d0',
      color: ambar,
      fontSize: '10px',
      fontWeight: 700,
      letterSpacing: '.06em',
      whiteSpace: 'nowrap',
    }}
  >
    EN CURSO
  </span>
)

export const Cifra = ({ n, que, color }) => (
  <div
    style={{
      flex: '1 1 130px',
      background: '#fff',
      border: `1px solid ${linea}`,
      borderRadius: '12px',
      padding: '16px 18px',
    }}
  >
    <div style={{ fontSize: '32px', fontWeight: 700, lineHeight: 1.1, color: color || tinta }}>
      {n}
    </div>
    <div style={{ fontSize: '13px', color: gris, marginTop: '2px' }}>{que}</div>
  </div>
)

export const Fila = ({ p, conDetalle }) => (
  <div
    style={{
      display: 'flex',
      alignItems: 'flex-start',
      gap: '10px',
      padding: '11px 0',
      borderBottom: `1px solid ${linea}`,
    }}
  >
    <Marca tipo={p.cambio} />
    <div style={{ flex: '1 1 auto', minWidth: 0 }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: '8px', flexWrap: 'wrap' }}>
        <span style={{ fontSize: '14px', color: tinta }}>{p.titulo}</span>
        {p.enCurso && !aPro ? <Pastilla /> : null}
      </div>
      {conDetalle && p.detalle?.length ? (
        <div style={{ fontSize: '12px', color: gris, marginTop: '4px', wordBreak: 'break-word' }}>
          {p.detalle.slice(0, 4).join(' · ')}
          {p.detalle.length > 4 ? ` · y ${p.detalle.length - 4} más` : ''}
        </div>
      ) : null}
    </div>
  </div>
)

export const Bloque = ({ titulo, que, lista, conDetalle, vacio }) => (
  <section style={{ marginBottom: '40px' }}>
    <h2 style={{ fontSize: '22px', margin: '0 0 4px', color: tinta }}>{titulo}</h2>
    <p style={{ margin: '0 0 14px', fontSize: '14px', color: gris, maxWidth: '70ch' }}>{que}</p>
    {lista.length ? (
      <div>
        {lista.map((p) => (
          <Fila key={p.titulo} p={p} conDetalle={conDetalle} />
        ))}
      </div>
    ) : (
      <div
        style={{
          padding: '14px 16px',
          borderRadius: '10px',
          background: suave,
          fontSize: '14px',
          color: gris,
        }}
      >
        {vacio}
      </div>
    )}
  </section>
)

  <Cifra n={resumen.nuevas ?? 0} que="piezas nuevas" color={verde} />
